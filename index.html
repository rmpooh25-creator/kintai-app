function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let passSheet = ss.getSheetByName("パスワード管理"); if (!passSheet) { passSheet = ss.insertSheet("パスワード管理"); passSheet.appendRow(["名前", "パスワード"]); return ContentService.createTextOutput(JSON.stringify({ names: [] })) .setMimeType(ContentService.MimeType.JSON); } const data = passSheet.getDataRange().getValues(); const names = []; for (let i = 1; i < data.length; i++) { if (data[i][0]) { names.push(String(data[i][0])); } } return ContentService.createTextOutput(JSON.stringify({ names: names })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const name = data.name; const password = data.password; const type = data.type; // "出勤" または "退勤" const customTime = data.customTime; // 手動入力時間 const note = data.note || ""; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. パスワード管理シートの確認・検証・新規登録 let passSheet = ss.getSheetByName("パスワード管理"); if (!passSheet) { passSheet = ss.insertSheet("パスワード管理"); passSheet.appendRow(["名前", "パスワード"]); } const passData = passSheet.getDataRange().getValues(); let userExists = false; let passwordMatched = false; for (let i = 1; i < passData.length; i++) { if (passData[i][0] === name) { userExists = true; if (String(passData[i][1]) === String(password)) { passwordMatched = true; } break; } } if (!userExists) { passSheet.appendRow([name, password]); } else if (!passwordMatched) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "パスワードが間違っています。" })).setMimeType(ContentService.MimeType.JSON); } // 2. 個人シートの取得・作成処理（テンプレート複製または新規作成） let userSheet = ss.getSheetByName(name); if (!userSheet) { const templateSheet = ss.getSheetByName("テンプレート"); if (templateSheet) { userSheet = templateSheet.copyTo(ss); userSheet.setName(name); } else { userSheet = ss.insertSheet(name); userSheet.appendRow(["日付", "打刻者", "出勤時間", "退勤時間", "実働時間", "備考"]); } } // 3. JST基準の日付と時間の取得 const now = new Date(); const targetYear = now.getFullYear(); const targetMonth = now.getMonth(); const targetDate = now.getDate(); const dateStr = Utilities.formatDate(now, "JST", "yyyy/MM/dd"); let timeStr = Utilities.formatDate(now, "JST", "HH:mm:ss"); if (customTime) { // 修正時間のフォーマット整合性を保持 timeStr = customTime.length === 5 ? customTime + ":00" : customTime; } const rows = userSheet.getDataRange().getValues(); // 日付が一致するか判定する安全なヘルパー関数 function isSameDate(cellValue) { if (!cellValue) return false; let d; if (cellValue instanceof Date) { d = cellValue; } else { d = new Date(cellValue); } if (isNaN(d.getTime())) { return String(cellValue).includes(dateStr); } return d.getFullYear() === targetYear && d.getMonth() === targetMonth && d.getDate() === targetDate; } // 本日の既存行を検索 (A列=日付, B列=打刻者, C列=出勤時間, D列=退勤時間, E列=実働時間, F列=備考) let todayRowIndex = -1; for (let i = 1; i < rows.length; i++) { if (isSameDate(rows[i][0])) { todayRowIndex = i + 1; // 1-indexedの行番号 break; } } if (type === "出勤") { if (todayRowIndex !== -1) { if (rows[todayRowIndex - 1][2] && !customTime) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "すでに本日の出勤打刻が完了しています。" })).setMimeType(ContentService.MimeType.JSON); } userSheet.getRange(todayRowIndex, 3).setValue(timeStr); let currentNote = rows[todayRowIndex - 1][5] || ""; if (customTime) { currentNote = currentNote ? currentNote + " 【出勤修正済】" : "【出勤修正済】"; } if (note) currentNote = currentNote ? currentNote + " " + note : note; userSheet.getRange(todayRowIndex, 6).setValue(currentNote); } else { const newRowIdx = rows.length + 1; const formula = `=IF(AND(C${newRowIdx}<>"", D${newRowIdx}<>""), D${newRowIdx}-C${newRowIdx}, "")`; const finalNote = customTime ? (note ? note + " 【出勤修正済】" : "【出勤修正済】") : note; userSheet.appendRow([dateStr, name, timeStr, "", formula, finalNote]); } } else if (type === "退勤") { // 未打刻チェック（ご要望のエラーメッセージ） if (todayRowIndex === -1 || !rows[todayRowIndex - 1][2]) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "エラー: 本日の出勤打刻がされていません。「出勤打刻を修正してから退勤してください。※一回のみ」" })).setMimeType(ContentService.MimeType.JSON); } if (rows[todayRowIndex - 1][3] && !customTime) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "すでに本日の退勤打刻が完了しています。" })).setMimeType(ContentService.MimeType.JSON); } userSheet.getRange(todayRowIndex, 4).setValue(timeStr); let currentNote = rows[todayRowIndex - 1][5] || ""; if (customTime && rows[todayRowIndex - 1][3]) { currentNote = currentNote ? currentNote + " 【退勤修正済】" : "【退勤修正済】"; } if (note) currentNote = currentNote ? currentNote + " " + note : note; userSheet.getRange(todayRowIndex, 6).setValue(currentNote); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "正常に記録されました。" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
