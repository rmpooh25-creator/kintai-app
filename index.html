function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. パスワード管理シートの取得（無ければ自動作成） let passSheet = ss.getSheetByName("パスワード管理"); if (!passSheet) { passSheet = ss.insertSheet("パスワード管理"); passSheet.appendRow(["名前", "パスワード"]); } const name = data.name ? data.name.trim() : ""; const password = data.password ? data.password.trim() : ""; const type = data.type; // 出勤 または 退勤 const note = data.note || ""; if (!name || !password) { return responseJSON({ status: "error", message: "名前とパスワードを入力してください。" }); } // 2. パスワード認証・新規ユーザー登録 const passData = passSheet.getDataRange().getValues(); let userRow = -1; for (let i = 1; i < passData.length; i++) { if (passData[i][0] === name) { userRow = i + 1; break; } } if (userRow === -1) { // 新規登録 passSheet.appendRow([name, password]); } else { // パスワードチェック const savedPass = passData[userRow - 1][1].toString(); if (savedPass !== password) { return responseJSON({ status: "error", message: "パスワードが正しくありません。" }); } } // 3. 個人用シートの取得（無ければ「テンプレート」を複製して自動作成） let userSheet = ss.getSheetByName(name); if (!userSheet) { const templateSheet = ss.getSheetByName("テンプレート"); if (templateSheet) { userSheet = templateSheet.copyTo(ss).setName(name); } else { userSheet = ss.insertSheet(name); userSheet.appendRow(["日時", "区分", "実働時間", "備考"]); } } // 4. 打刻日時と実働時間の計算 const now = new Date(); let workTimeStr = ""; if (type === "退勤") { const userData = userSheet.getDataRange().getValues(); const todayStr = Utilities.formatDate(now, "Asia/Tokyo", "yyyy/MM/dd"); let lastInTime = null; for (let i = userData.length - 1; i >= 1; i--) { const rowDate = userData[i][0]; const rowType = userData[i][1]; if (rowType === "出勤" && rowDate instanceof Date) { const rowDateStr = Utilities.formatDate(rowDate, "Asia/Tokyo", "yyyy/MM/dd"); if (rowDateStr === todayStr) { lastInTime = rowDate; break; } } } if (lastInTime) { const diffMs = now.getTime() - lastInTime.getTime(); const totalMinutes = Math.floor(diffMs / (1000 * 60)); const hours = Math.floor(totalMinutes / 60); const minutes = totalMinutes % 60; workTimeStr = hours + "時間" + minutes + "分"; } } // 5. 個人用シートの末尾に書き込み userSheet.appendRow([now, type, workTimeStr, note]); return responseJSON({ status: "success", message: "【" + type + "】を受け付けました！" + (workTimeStr ? "（本日の勤務時間: " + workTimeStr + "）" : "") }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function responseJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
